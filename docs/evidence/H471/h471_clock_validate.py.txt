if suite=='normal':
 for cc in ['g++','clang++']:run(['bash',str(r/'tests/run_host_tests.sh')],env={**os.environ,'CXX':cc,'EDP_SYSTEM_SOURCE_DIR':str(workspace/'EDP-System'),'EDP_PLATFORM_SOURCE_DIR':str(workspace/'EDP-Platform')})

  for source in sources:
   name=source.stem;flags=['-std=c++20','-Wall','-Wextra','-Werror','-pedantic','-pthread','-I'+str(r/'src'),'-I'+str(workspace/'EDP-System/src'),'-I'+str(workspace/'EDP-Platform/src')]

    flags+=['--analyze','-Xanalyzer','-analyzer-werror','-Xanalyzer','-analyzer-output=text']
